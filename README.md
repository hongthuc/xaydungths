    function handleDriveImageError(imgEl, driveId, fallbackLabel) {
      const step = parseInt(imgEl.getAttribute('data-retry-step') || '0', 10);
      if (step === 0) {
        imgEl.setAttribute('data-retry-step', '1');
        imgEl.src = 'https://drive.google.com/thumbnail?id=' + driveId + '&sz=w1000';
      } else if (step === 1) {
        imgEl.setAttribute('data-retry-step', '2');
        imgEl.src = 'https://drive.google.com/uc?export=view&id=' + driveId;
      } else {
        imgEl.onerror = null;
        const encodedText = encodeURIComponent(fallbackLabel || 'XAY DUNG THS');
        imgEl.src = 'https://placehold.co/800x600/0f3b68/ffffff?text=' + encodedText;
      }
    }

        }
      });
    }
